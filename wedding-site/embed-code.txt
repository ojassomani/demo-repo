/* Arch frames are redrawn to each card's real shape, so the arch never looks stretched */
function drawFrames() {
  document.querySelectorAll("[data-frame]").forEach(el => {
    const r = el.getBoundingClientRect();
    if (!r.width) return;
    const H = Math.max(300, Math.round(300 * r.height / r.width));
    if (el.dataset.h === String(H)) return;
    el.dataset.h = H;
    el.setAttribute("viewBox", `0 0 300 ${H}`);
    el.innerHTML = frame(el.dataset.frame, H);
  });
}

/* ================= Countdown to the Shaadi: 28 Dec 2026, 11:00 AM IST (05:30 UTC) ================= */
(function () {
  const target = Date.UTC(2026, 11, 28, 5, 30, 0);
  const box = document.getElementById("countdown");
  const d = document.getElementById("cd-d"), h = document.getElementById("cd-h"), m = document.getElementById("cd-m");
  function tick() {
    const diff = target - Date.now();
    if (diff <= 0) {
      box.innerHTML = '<div class="cd-done">Married on 28 December 2026</div>';
      return false;
    }
    const mins = Math.floor(diff / 60000);
    d.textContent = Math.floor(mins / 1440);
    h.textContent = String(Math.floor((mins % 1440) / 60)).padStart(2, "0");
    m.textContent = String(mins % 60).padStart(2, "0");
    return true;
  }
  if (tick()) { const id = setInterval(() => { if (!tick()) clearInterval(id); }, 30000); }
})();

/* ================= Copy phone number, with a fallback that selects the number ================= */
document.querySelectorAll(".copy").forEach(btn => {
  btn.addEventListener("click", () => {
    const label = btn.textContent;
    const done = () => { btn.textContent = "Copied"; setTimeout(() => { btn.textContent = label; }, 1600); };
    const fallback = () => {
      const el = document.getElementById(btn.dataset.target);
      const r = document.createRange(); r.selectNodeContents(el);
      const s = window.getSelection(); s.removeAllRanges(); s.addRange(r);
      btn.textContent = "Number selected";
      setTimeout(() => { btn.textContent = label; }, 1600);
    };
    try {
      navigator.clipboard.writeText(btn.dataset.num).then(done, fallback);
    } catch (e) { fallback(); }
  });
});
